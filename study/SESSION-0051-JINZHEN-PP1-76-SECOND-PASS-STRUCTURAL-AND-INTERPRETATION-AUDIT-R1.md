from collections import Counter
B='子丑寅卯辰巳午未申酉戌亥'; S='甲乙丙丁戊己庚辛壬癸'
L='寅辰巳未巳未申戌亥丑'
se=[0,0,1,1,2,2,3,3,4,4]; be=[4,2,0,0,2,1,1,2,3,3,2,4]
k=lambda x,y:(x+2)%5==y
counts=Counter(); fanyin=[]; all_unmatched=[]; ties=[]
for i in range(60):
    s=i%10; b=i%12; a=B.index(L[s])
    for d in range(12):
        u=[(a+d)%12,(a+2*d)%12,(b+d)%12,(b+2*d)%12]
        lo=[se[s],be[u[0]],be[b],be[u[2]]]; up=[be[x] for x in u]
        counts[len(set(u))]+=1
        down=[v for v,x,y in zip(u,lo,up) if k(x,y)]
        above=[v for v,x,y in zip(u,lo,up) if k(y,x)]
        can=set(down or above)
        if d==6 and not can: fanyin.append(S[s]+B[b])
        if d in (0,6) or len(can)<=1: continue
        same={v for v in can if v%2==s%2}
        if not same: all_unmatched.append((S[s]+B[b],d,''.join(B[v] for v in sorted(can))))
        pool=same or can  # diagnostic fallback only, not settled historical rule
        if len(pool)>1:
            rank=lambda v:((v-d)%12-2)%3
            best={v for v in pool if rank(v)==min(map(rank,pool))}
            if len(best)>1: ties.append((S[s]+B[b],d,''.join(B[v] for v in sorted(best))))
print(counts,fanyin,all_unmatched,ties)
